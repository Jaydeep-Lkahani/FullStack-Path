## Stage 3: Sessions, `requireAuth`, login, `/me` and logout

Now make the login stick. Write:

- `src/sessions.ts`: the cookie name (`sid`), the lifetime (7 days), a `Map` from session id to `{ userId, expiresAt }`, and the cookie options (`httpOnly`, `sameSite: "lax"`, `secure` when `NODE_ENV` is `production`, `path: "/"`). `createSession(userId)` makes the id with `randomBytes(32)` in `base64url`. `getSession(id)` returns nothing for an unknown **or expired** session and deletes the expired one. Add `destroySession(id)` and `resetSessions()` for tests.
- `src/express.d.ts`: teaches TypeScript that `req.user` and `req.sessionId` exist after `requireAuth` ran.
- `src/middleware.ts` with `requireAuth` and a helper `currentAuth(req)`. `requireAuth` reads the cookie (it can be anything the client sends, so check that it is a string), finds the session, then finds the **user** and answers 401 when any step fails. It looks the user up on every request on purpose. `currentAuth` returns `{ user, sessionId }` or throws 401, so handlers need no `!`.
- The other routes in `src/routes/auth.ts`: `POST /login` (parse, `checkLogin`, 401 "Invalid email or password", end the old session if the request had a cookie, create a new one, set the cookie with `maxAge`, answer the user), `GET /me`, and `POST /logout` (destroy the session on the server, clear the cookie with the **same options**, answer 204).
- In `app.ts`, add `cookieParser()` before the routes.

<Solution>

```ts
// src/sessions.ts
import { randomBytes } from "node:crypto";
import type { CookieOptions } from "express";
import { config } from "./config.ts";

export const SESSION_COOKIE = "sid";
export const SESSION_TTL_MS = 7 * 24 * 60 * 60 * 1000; // one week

type Session = { userId: string; expiresAt: number };

// session id -> who it belongs to. In memory, so a restart logs everyone out.
const sessions = new Map<string, Session>();

export const cookieOptions: CookieOptions = {
  httpOnly: true, // JavaScript in the page cannot read it
  sameSite: "lax", // not sent on cross-site POSTs
  secure: config.NODE_ENV === "production", // HTTPS only in production
  path: "/",
};

// for tests: no sessions again
export function resetSessions(): void {
  sessions.clear();
}

export function createSession(userId: string): string {
  const id = randomBytes(32).toString("base64url"); // 256 random bits: cannot be guessed
  sessions.set(id, { userId, expiresAt: Date.now() + SESSION_TTL_MS });
  return id;
}

export function getSession(id: string): Session | undefined {
  const session = sessions.get(id);
  if (!session) return undefined;
  if (session.expiresAt <= Date.now()) {
    sessions.delete(id);
    return undefined;
  }
  return session;
}

export function destroySession(id: string): void {
  sessions.delete(id);
}
```

```ts
// src/middleware.ts
import type { NextFunction, Request, Response } from "express";
import { unauthorized } from "./errors.ts";
import { getSession, SESSION_COOKIE } from "./sessions.ts";
import { findUserById, toPublic } from "./users.ts";
import type { PublicUser } from "./users.ts";

// 401: "who are you?" The user is looked up on every request, so a changed role,
// a changed password or a deleted session takes effect at once.
export function requireAuth(req: Request, res: Response, next: NextFunction) {
  const cookie: unknown = req.cookies?.[SESSION_COOKIE];
  const sessionId = typeof cookie === "string" ? cookie : undefined;
  const session = sessionId && getSession(sessionId);
  const user = session && findUserById(session.userId);

  if (!sessionId || !user) {
    next(unauthorized());
    return;
  }

  req.user = toPublic(user);
  req.sessionId = sessionId;
  next();
}

// for handlers behind requireAuth: the user and the session id, typed, without a `!`
export function currentAuth(req: Request): { user: PublicUser; sessionId: string } {
  if (!req.user || !req.sessionId) throw unauthorized();
  return { user: req.user, sessionId: req.sessionId };
}
```

```ts
// src/routes/auth.ts
import { Router } from "express";
import { unauthorized } from "../errors.ts";
import { currentAuth, requireAuth } from "../middleware.ts";
import { loginSchema, signupSchema } from "../schemas.ts";
import {
  cookieOptions,
  createSession,
  destroySession,
  SESSION_COOKIE,
  SESSION_TTL_MS,
} from "../sessions.ts";
import { checkLogin, createUser, toPublic } from "../users.ts";
import { parse } from "../validate.ts";

export const authRouter = Router();

// POST /v1/auth/signup  { "email": "...", "password": "..." }
authRouter.post("/signup", async (req, res) => {
  const { email, password } = parse(signupSchema, req.body, "body");
  const user = await createUser(email, password); // role is always "user": the body cannot say otherwise
  res
    .status(201)
    .location("/v1/auth/me")
    .json({ data: toPublic(user) });
});

// POST /v1/auth/login  { "email": "...", "password": "..." }
authRouter.post("/login", async (req, res) => {
  const { email, password } = parse(loginSchema, req.body, "body");
  const user = await checkLogin(email, password);
  if (!user) throw unauthorized("Invalid email or password"); // the same text for both causes

  // a new id for every login, and the old one dies: nobody can hand you a session id in advance
  const old: unknown = req.cookies?.[SESSION_COOKIE];
  if (typeof old === "string") destroySession(old);

  res.cookie(SESSION_COOKIE, createSession(user.id), {
    ...cookieOptions,
    maxAge: SESSION_TTL_MS,
  });
  res.json({ data: toPublic(user) });
});

// GET /v1/auth/me
authRouter.get("/me", requireAuth, (req, res) => {
  res.json({ data: currentAuth(req).user });
});

// POST /v1/auth/logout: removes the session on the server, then the cookie in the browser
authRouter.post("/logout", requireAuth, (req, res) => {
  destroySession(currentAuth(req).sessionId);
  res.clearCookie(SESSION_COOKIE, cookieOptions);
  res.status(204).end();
});
```

```text
$ curl -b jar.txt localhost:4000/v1/auth/me

$ curl -b old-jar.txt localhost:4000/v1/auth/me
```

The last request is the point of server-side sessions: the copied cookie is still well-formed, but the session behind it no longer exists, so it is useless. With a JWT you could not do this before the token expired.

Two details to notice in the code. The `login` route destroys the session id the browser sent before it creates a new one, so a session id that someone planted in a browser before the login (**session fixation**) never becomes a logged-in session. And `res.clearCookie` gets `cookieOptions` again, because a browser only removes a cookie when the name, the path and the flags match.

## Stage 4: Roles, the admin routes and the first admin

Add roles to the service. The rules:

- `GET /v1/users` lists all users (without hashes) and is for admins only. A stranger gets **401**, a logged-in normal user gets **403**.
- `PATCH /v1/users/:id` with `{ "role": "admin" }` or `{ "role": "user" }` changes a role. A bad id is a 400, an unknown user a 404, a bad role a 400. **Nobody can change their own role** (403), because otherwise the last admin could demote themselves and nobody could undo it.
- The new role works on the **next request** with the same cookie. That happens by itself, because `requireAuth` reads the user from the store every time. This is why a session stores a user id and not a copy of the user.
- Someone has to be the first admin. At startup, if `ADMIN_EMAIL` and `ADMIN_PASSWORD` are set and no such user exists, create it with the role `admin`. This happens in `server.ts`, not in `app.ts`, so tests never create it.

You need: `requireRole(...roles)` in `middleware.ts`, two schemas, two store functions, `src/routes/users.ts` (one `usersRouter.use(requireAuth, requireRole("admin"))` protects the whole file), `src/seedAdmin.ts`, and two lines each in `app.ts` and `server.ts`.

```ts
// 403: "you may not". Use it after requireAuth.
export function requireRole(...allowed: Role[]) {
  return (req: Request, res: Response, next: NextFunction) => {
    if (!req.user) {
      next(unauthorized());
    } else if (!allowed.includes(req.user.role)) {
      next(forbidden());
    } else {
      next();
    }
  };
}
```

```ts
import { forbidden, notFound } from "../errors.ts";
import { currentAuth, requireAuth, requireRole } from "../middleware.ts";

import { listUsers, setRole, toPublic } from "../users.ts";
```

## Stage 5: Change password and log out everywhere

Two routes for when credentials may be in the wrong hands.

- `POST /v1/auth/change-password` with `{ "currentPassword", "newPassword" }`, for a logged-in user. The new password follows the sign-up rules and must differ from the current one (a 400). A wrong current password is a **403** with the code `WRONG_PASSWORD`, not a 401: the client is logged in, and a 401 would make a front end throw the user back to the login page. On success: hash and store the new password, **destroy every other session of the user but keep the current one**, answer 204. Think about why: if the password changes because someone else knew it, that someone may be logged in right now.
- `POST /v1/auth/logout-all`: destroys every session of the user, this one too, clears the cookie, answers 204.

You need two store functions (`verifyPassword`, `changePassword`), one in the session store (`destroyUserSessions(userId, keepId?)`), one schema and the two routes.

<Solution>

In `src/schemas.ts`:

```ts
// src/schemas.ts (added)
export const changePasswordSchema = z
  .object({
    currentPassword: z.string({ error: "currentPassword is required" }).min(1).max(128),
    newPassword: newPassword("newPassword"),

  .refine((body) => body.currentPassword !== body.newPassword, {
    path: ["newPassword"],
```

```ts
// src/sessions.ts (added, at the end)
// every session of one user, except the one in `keepId` (leave it out to end them all)
export function destroyUserSessions(userId: string, keepId?: string): void {
  for (const [id, session] of sessions) {
    if (session.userId === userId && id !== keepId) sessions.delete(id);
  }
}
```

```ts
// src/routes/auth.ts (added)
// POST /v1/auth/logout-all: ends this session and every other one of the same user
authRouter.post("/logout-all", requireAuth, (req, res) => {
  destroyUserSessions(currentAuth(req).user.id);
  res.clearCookie(SESSION_COOKIE, cookieOptions);
  res.status(204).end();
});

// POST /v1/auth/change-password  { "currentPassword": "...", "newPassword": "..." }
authRouter.post("/change-password", requireAuth, async (req, res) => {
  const { user: publicUser, sessionId } = currentAuth(req);
  const { currentPassword, newPassword } = parse(changePasswordSchema, req.body, "body");

  const user = findUserById(publicUser.id);
  if (!user) throw notFound("No such user");
  if (!(await verifyPassword(user, currentPassword))) {
    // 403, not 401: the client is logged in, a 401 would make it show the login page
    throw new HttpError(403, "WRONG_PASSWORD", "The current password is wrong");
  }

  await changePassword(user, newPassword);
  destroyUserSessions(user.id, sessionId); // a thief who had the old password loses their session
  res.status(204).end();
});
```

<Callout type="warning" title="This route is a password-guessing machine">
Anyone holding a stolen session cookie can try passwords against `change-password` as fast as the server hashes them. In a real service the `security` lesson's rate limiter belongs on this route too, together with the login. It is a stretch goal here, so the suite stays about the rules and not about the limiter.
</Callout>
